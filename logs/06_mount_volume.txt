### B. Docker 볼륨 영속성
$ docker volume create mydata
mydata

$ docker run -d --name vol-test -v mydata:/data ubuntu sleep infinity
2c336afb31c5eb188bd39747a571c1419eaf3c005789d6902314d608c835933d

$ docker exec vol-test bash -c echo 'hi from volume' > /data/hello.txt && cat /data/hello.txt
hi from volume

$ docker volume ls
DRIVER    VOLUME NAME
local     mydata

### 컨테이너 삭제 (볼륨은 남음)
$ docker rm -f vol-test
vol-test

$ docker ps -a --filter name=vol-test
CONTAINER ID   IMAGE     COMMAND   CREATED   STATUS    PORTS     NAMES

$ docker volume ls
DRIVER    VOLUME NAME
local     mydata

### 새 컨테이너에 같은 볼륨 연결 후 데이터 확인
$ docker run -d --name vol-test2 -v mydata:/data ubuntu sleep infinity
be786e5d0b68210a6db86f5e2d4ce715aec6da64c4530f6f138585af68096dbc

$ docker exec vol-test2 cat /data/hello.txt
hi from volume

### 대조: 볼륨 없이 쓴 데이터는 컨테이너와 함께 사라짐
$ docker run -d --name no-vol ubuntu sleep infinity
92768b5b73af6a24dacc4acbca7bb54d6f44d5ebfbd6b61a2663e39d4f157a74

$ docker exec no-vol bash -c echo temp > /tmp/temp.txt && cat /tmp/temp.txt
temp

$ docker rm -f no-vol
no-vol

$ docker run --rm ubuntu bash -c ls /tmp/temp.txt
ls: cannot access '/tmp/temp.txt': No such file or directory
